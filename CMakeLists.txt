cmake_minimum_required(VERSION 3.24)
project(MapExtractor VERSION 0.8.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
add_executable(MapExtractor src/main.cpp src/ui.cpp src/doors.cpp src/water.cpp src/geodata.cpp src/collision.cpp src/localization.cpp src/history.cpp resources/MapExtractor.rc)
target_include_directories(MapExtractor PRIVATE resources)
target_compile_definitions(MapExtractor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(MapExtractor PRIVATE Threads::Threads comdlg32 shell32 dwmapi ole32 uuid comctl32 uxtheme gdiplus)
# Windows subsystem: opening the exe does not create a console. wmain stays the entry point.
set_target_properties(MapExtractor PROPERTIES WIN32_EXECUTABLE TRUE)
if(MSVC)
    target_compile_options(MapExtractor PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_link_options(MapExtractor PRIVATE "/ENTRY:wmainCRTStartup" "/MANIFEST:NO")
endif()
install(TARGETS MapExtractor RUNTIME DESTINATION .)
install(DIRECTORY profiles DESTINATION .)
install(DIRECTORY Languages DESTINATION .)
